// ==========================================
// LEAFLET MAP SUB-COMPONENT (DRAGGABLE BY DEFAULT)
// ==========================================
function LeafletMapContainer({ landmarks, selectedLandmark, onSelectLandmark, lang }) {
  const mapRef = useRef(null);
  const leafletInstance = useRef(null);
  const markersRef = useRef({});

  useEffect(() => {
    if (!mapRef.current || !window.L) return;

    // Initialize Leaflet Map centered over Hong Kong
    if (!leafletInstance.current) {
      const map = window.L.map(mapRef.current, {
        center: [22.3193, 114.1694],
        zoom: 11,
        zoomControl: true,
        dragging: true, // Directly draggable
        scrollWheelZoom: true,
        touchZoom: true
      });

      window.L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '&copy; OpenStreetMap contributors'
      }).addTo(map);

      leafletInstance.current = map;
    }

    const map = leafletInstance.current;

    // Clear existing markers
    Object.values(markersRef.current).forEach(m => m.remove());
    markersRef.current = {};

    // Custom Icon Generator for 18 Districts
    landmarks.forEach(lm => {
      const isSelected = selectedLandmark?.id === lm.id;
      
      const customHtml = `
        <div style="
          background-color: ${isSelected ? '#f59e0b' : '#0d9488'};
          color: white;
          width: 34px;
          height: 34px;
          border-radius: 50%;
          display: flex;
          align-items: center;
          justify-content: center;
          border: 3px solid white;
          box-shadow: 0 4px 10px rgba(0,0,0,0.3);
          transform: ${isSelected ? 'scale(1.2)' : 'scale(1)'};
          transition: transform 0.2s;
        ">
          <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round">
            <path d="M20 10c0 6-8 12-8 12s-8-6-8-12a8 8 0 0 1 16 0Z"></path>
            <circle cx="12" cy="10" r="3"></circle>
          </svg>
        </div>
      `;

      const customIcon = window.L.divIcon({
        className: 'custom-leaflet-pin',
        html: customHtml,
        iconSize: [34, 34],
        iconAnchor: [17, 34]
      });

      const marker = window.L.marker([lm.lat, lm.lng], { icon: customIcon })
        .addTo(map)
        .on('click', () => {
          onSelectLandmark(lm);
        });

      // Bind tooltip popup
      marker.bindTooltip(`<b>${lm.nameZh}</b><br/><span style="font-size:10px;">${lm.districtZh}</span>`, {
        direction: 'top',
        offset: [0, -30]
      });

      markersRef.current[lm.id] = marker;
    });

  }, [landmarks, selectedLandmark]);

  // Pan to landmark when selected
  useEffect(() => {
    if (selectedLandmark && leafletInstance.current) {
      leafletInstance.current.flyTo([selectedLandmark.lat, selectedLandmark.lng], 15, {
        duration: 1.2
      });
    }
  }, [selectedLandmark]);

  return (
    <div className="relative w-full h-full">
      <div ref={mapRef} className="w-full h-full" />
    </div>
  );
}
